# Recommended mapping (matches typical #status-reports / #errors / #support webhooks):
#   SLACK_WEBHOOK_STATUS_REPORTS — cron/manual sync digests (routine runs)
#   SLACK_WEBHOOK_ERRORS — failed sync steps, health-check failures
#   SLACK_WEBHOOK_SUPPORT — optional copy of *critical* only: sync crashes + health failures
#                             (does not receive per-source row failures; those stay on ERRORS)

# Legacy names still work: SLACK_WEBHOOK_SYNC (digest), SLACK_WEBHOOK_ALERTS (errors),
# SLACK_WEBHOOK_URL (fallback for digest then errors chain).
#
# Digest posts for Vercel Cron syncs; set SLACK_SYNC_NOTIFY_MANUAL=true to also post manual runs.
# CLI / GitHub Actions (`tsx scripts/manual-sync.ts`): set SLACK_SYNC_NOTIFY_CLI=true (see workflows).
# By default, status-reports digest only when something synced (itemsSynced > 0) or there were errors/skips.
# Set SLACK_SYNC_DIGEST_ALWAYS=true for a digest on every Vercel cron run (all sources). Recommended in Vercel
#   for once-daily visibility that legislators / votes / lrc-calendar actually ran (they are otherwise silent).

# Content accuracy audit (npm run audit:accuracy — weekly via
# .github/workflows/accuracy-audit.yml). Re-fetches LegiScan / Open States / LRC,
# diffs against Supabase, runs an Anthropic LLM pass over fuzzy content, posts to
# the same Slack webhooks (gated by SLACK_SYNC_NOTIFY_CLI=true from CI). Defaults
# shown; all optional.
# ACCURACY_SEED=                        — fix the random sample seed for reproducible runs (default: random each run; printed on every run).
# ACCURACY_DAYS=14                      — reserved for time-windowed checks (committees use live data; bills/votes now seed-sample).
# ACCURACY_BILLS_LIMIT=40               — max bills re-fetched from LegiScan per run.
# ACCURACY_VOTES_LIMIT=15               — max roll calls re-fetched from LegiScan per run.
# ACCURACY_MATERIALS_COMMITTEE_LIMIT=12 — max committees re-scraped for materials per run.
# ACCURACY_LINK_SAMPLE=25               — max stored URLs checked per run.
# ACCURACY_PROBE_LINKS=false            — false: validate URLs statically vs canonical hosts/shape (fast, no network).
#                                         true: live HTTP reachability probes (slower; concurrency-limited with retries).
# ACCURACY_LLM_SAMPLE=8                 — sample size per LLM review pass.
# ACCURACY_SKIP_LLM=false               — set true (or --no-llm) to skip the Anthropic pass.
# ACCURACY_LLM_MODEL=claude-sonnet-4-6
# ACCURACY_LEGISCAN_QUOTA_STOP_PCT=95   — skip LegiScan-backed checks at/above this monthly usage %.
